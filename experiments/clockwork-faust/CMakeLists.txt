cmake_minimum_required(VERSION 3.24)
project(AliveClockworkProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CLOCKWORK_DSP ultra_research CACHE STRING "" FORCE)
set(CLOCKWORK_DSP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/guest" CACHE PATH "" FORCE)
foreach(feature DEVICE MIDI GAMEPAD OSC PLUGINS LINK STANDALONE)
 set(CLOCKWORK_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(CLOCKWORK_SCHEDULER ON CACHE BOOL "" FORCE)
set(CLOCKWORK_COMMIT "94b67500c96e105491682ec68741b5018560eab7")
add_subdirectory(.deps/clockwork-94b67500c96e105491682ec68741b5018560eab7 clockwork)
add_executable(probe probe.cpp instrument.cpp)
target_link_options(probe PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free,--wrap=pthread_mutex_lock)
target_link_libraries(probe PRIVATE clockwork_bundle)
target_include_directories(probe SYSTEM PRIVATE ../faust-reference/.deps/faust-2.88.0/architecture)
